function login() { const id = document.getElementById('userId').value.trim(); const name = document.getElementById('userName').value.trim(); const role = document.getElementById('role').value.trim().toLowerCase(); const message = document.getElementById('loginMessage'); if (!id || !name) { message.textContent = 'Please enter your ID and full name.'; return; } if (role !== 'staff' && role !== 'student') { message.textContent = 'Please select Staff or Student.'; return; } sessionStorage.setItem('attendanceUser', name); sessionStorage.setItem('attendanceRole', role); sessionStorage.setItem('attendanceUserId', id); window.location.href = './dashboard.html'; }
