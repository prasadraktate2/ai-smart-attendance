function login() { const id = document.getElementById('userId').value.trim(); const name = document.getElementById('userName').value.trim(); const role = document.getElementById('role').value; if (!id || !name) { document.getElementById('loginMessage').textContent = 'Please enter your ID and full name.'; return; } currentUser = { id, name, role }; // Share login details with dashboard.html sessionStorage.setItem('attendanceUser', name); sessionStorage.setItem('attendanceRole', role); sessionStorage.setItem('attendanceUserId', id); window.location.href = 'dashboard.html'; }
